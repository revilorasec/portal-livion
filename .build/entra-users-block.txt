function isExternalType(t){return t==='CLIENTE'||t==='PARCEIRO'}
function entraLabel(u){if(!isExternalType(u.user_type))return 'Interno';const s=String(u.entra_invite_status||'PENDING').toUpperCase();return s==='ACCEPTED'?'Aceito':s==='INVITED'?'Convite enviado':s==='ERROR'?'Erro no convite':'Pendente'}
async function acquireInviteToken(){const account=msalApp.getActiveAccount();if(!account)throw new Error('Sessão Microsoft não encontrada');try{return (await msalApp.acquireTokenSilent({scopes:[GRAPH_INVITE_SCOPE],account})).accessToken}catch(e){return (await msalApp.acquireTokenPopup({scopes:[GRAPH_INVITE_SCOPE],account,prompt:'consent'})).accessToken}}
async function sendEntraInvitation(email,name,graphToken){const r=await fetch('https://graph.microsoft.com/v1.0/invitations',{method:'POST',headers:{Authorization:'Bearer '+graphToken,'Content-Type':'application/json'},body:JSON.stringify({invitedUserEmailAddress:email,invitedUserDisplayName:name,inviteRedirectUrl:REDIRECT,sendInvitationMessage:true})});const b=await r.json().catch(()=>({}));if(!r.ok)throw new Error(b.error?.message||'Não foi possível enviar o convite Microsoft');const guestId=b.invitedUser?.id||'';await api('/users/entra-status',{method:'POST',body:JSON.stringify({email,status:'INVITED',guest_id:guestId})});return b}

function collect(){return {name:$('fName').value.trim(),email:$('fEmail').value.trim().toLowerCase(),profile:$('fProfile').value,user_type:$('fUserType').value,organization_key:$('fOrg')?.value.trim()||'',client_key:$('fClientKey')?.value.trim()||'',active:$('fActive').checked,apps:[...document.querySelectorAll('[data-app]:checked')].map(x=>x.dataset.app),companies:[...document.querySelectorAll('[data-company]:checked')].map(x=>x.dataset.company),actions:[...document.querySelectorAll('[data-action]:checked')].map(x=>x.dataset.action)}}
async function saveUser(){const notice=$('notice');try{const body=collect();const wasExisting=!!(editing&&users.some(u=>u.email===editing.email));let graphToken=null;if(isExternalType(body.user_type)&&!wasExisting){notice.textContent='Solicitando autorização Microsoft para enviar o convite...';graphToken=await acquireInviteToken()}await api('/users',{method:'POST',body:JSON.stringify(body)});let inviteMessage='';if(isExternalType(body.user_type)&&!wasExisting){try{await sendEntraInvitation(body.email,body.name,graphToken);inviteMessage='Usuário cadastrado e convite Microsoft enviado.'}catch(e){await api('/users/entra-status',{method:'POST',body:JSON.stringify({email:body.email,status:'ERROR'})}).catch(()=>{});inviteMessage='Usuário cadastrado, mas o convite Microsoft falhou: '+e.message}}users=(await api('/users')).users||[];editing=null;renderUsers();$('notice').textContent=inviteMessage||'Usuário salvo.'}catch(e){notice.textContent=e.message}}
async function reinviteUser(email){const u=users.find(x=>x.email===email);if(!u)return;try{const graphToken=await acquireInviteToken();await sendEntraInvitation(u.email,u.name,graphToken);users=(await api('/users')).users||[];renderUsers();$('notice').textContent='Convite Microsoft reenviado para '+u.email+'.'}catch(e){await api('/users/entra-status',{method:'POST',body:JSON.stringify({email,status:'ERROR'})}).catch(()=>{});alert('Não foi possível reenviar o convite: '+e.message)}}
async function toggleUser(email){const u=users.find(x=>x.email===email);if(!u)return;await api('/users',{method:'POST',body:JSON.stringify({name:u.name,email:u.email,profile:u.profile,user_type:u.user_type||'INTERNO',organization_key:u.organization_key||'',client_key:u.client_key||'',active:!u.active,apps:parse(u.apps_json),companies:parse(u.companies_json),actions:parse(u.actions_json)})});users=(await api('/users')).users||[];renderUsers()}
async function deleteUser(email){if(!confirm('Excluir '+email+'? O histórico de auditoria será preservado.'))return;await api('/users?email='+encodeURIComponent(email),{method:'DELETE'});users=(await api('/users')).users||[];editing=null;renderUsers()}
